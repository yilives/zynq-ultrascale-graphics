cmake_minimum_required(VERSION 3.16)
project(AXU3EGGameGL LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
include("${RIX_PROJECT_ROOT}/generated/rasterix_config.cmake")
set(RIX_BUILD_SHARED_LIBRARY OFF CACHE BOOL "Use the isolated GLX target" FORCE)
set(RIX_BUILD_ZYNQ_EMBEDDED_LINUX ON CACHE BOOL "AArch64 board" FORCE)
set(RIX_DRIVER_DMA_PROXY ON CACHE BOOL "Real FPGA bus" FORCE)
set(RIX_ENABLE_SPDLOG ON CACHE BOOL "Keep diagnostics" FORCE)
add_subdirectory("${RIX_PORT_CORE}" RasterIX)
add_library(GL SHARED "${RIX_PORT_CORE}/lib/glx/glx.cpp")
target_include_directories(GL PRIVATE "${RIX_PORT_CORE}/lib/glx")
target_link_libraries(GL PRIVATE -Wl,--whole-archive gl -Wl,--no-whole-archive
    dmaproxy utils span threadrunner spdlog::spdlog Threads::Threads X11)
set_target_properties(GL PROPERTIES VERSION 1.0 SOVERSION 1)
install(TARGETS GL LIBRARY DESTINATION lib)
